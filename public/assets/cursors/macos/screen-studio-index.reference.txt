const assetNameNsNameLookup = swapObjectKeysAndValues(nsNameAssetNameLookup);

const size: Vec = { x: 35, y: 35 };
const center: Vec = { x: 17.5, y: 17.5 };

const cursors = cursorAssetNames.map((cursorAssetName): CursorHighQualityAsset => {
  return {
    name: cursorAssetName,
    nsName: assetNameNsNameLookup[cursorAssetName] ?? null,
    filename: `${cursorAssetName}.svg`,
    hotspot: vec(customHotspots[cursorAssetName] ?? center),
    url: `screen-studio-asset://cursors/macos/png/${cursorAssetName}.png`,
    size,
  };
});

const nsNameCursorAssetLookup: Record<NsCursorName, CursorHighQualityAsset> = objectMap(
  nsNameAssetNameLookup,
  (nsName) => {
    return cursors.find((cursor) => cursor.name === nsName)!;
  },
);

export function getCursorAssetNameByNsName(nsName: NsCursorName): CursorAssetName | null {
  if (!getIsCursorNsName(nsName)) {
    return null;
  }

  return nsNameAssetNameLookup[nsName] ?? null;
}

export function getCursorAssetByNsName(nsName: NsCursorName): CursorHighQualityAsset | null {
  if (!getIsCursorNsName(nsName)) {
    return null;
  }

  return nsNameCursorAssetLookup[nsName] ?? null;
}

export function getIsCursorAssetName(name: string): name is CursorAssetName {
  return cursorAssetNames.includes(name as CursorAssetName);
}
